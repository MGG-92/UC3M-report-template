% Block diagrams package configuration (Tikz)
\usetikzlibrary{shapes, arrows, positioning, decorations.markings, calc}

\newcommand{\midof}[2]{($(#1)!0.5!(#2)$)}

\newlength{\dnd} % default node distance
\setlength{\dnd}{1cm}


\tikzset{
    every picture/.style={
        node distance=\dnd,
    },
    input/.style={
        inner ysep=0.75em,
        inner xsep=0.3em,
    },
    output/.style={
        inner ysep=0.75em,
        inner xsep=0.3em,
    },
    block/.style={
        draw, rectangle, 
        inner ysep=0.75em, 
        inner xsep=1em,
    },
    pickoff/.style={
        circle,
        fill=black,
        minimum size=4pt,
        inner sep=0pt,
    },
    sum/.style={
        draw,
        circle,
        minimum size=1.5em,
        inner sep=0pt,
    },
    plus/.style={
        pos=1,
        inner sep=2pt,
        font=\small,
        node contents={$+$}
    },
    minus/.style={
        pos=1,
        inner sep=2pt,
        font=\small,
        node contents={$-$}
    }
}
