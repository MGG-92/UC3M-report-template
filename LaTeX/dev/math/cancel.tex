\packageentry{cancel}{cancel -- Cancellations}
\partlabel{Preamble}
\begin{referencecode}
\usepackage{amsmath,cancel}
\end{referencecode}
\partlabel{Common factors}
\begin{referencecode}
\[
    \frac{\cancel{x}y}{\cancel{x}z}
    = \frac{y}{z},
    \qquad x \ne 0,\; z \ne 0
\]
\end{referencecode}

\[
    \frac{\cancel{x}y}{\cancel{x}z}
    = \frac{y}{z},
    \qquad x \ne 0,\; z \ne 0
\]

\partlabel{Cancel towards a value}
\begin{referencecode}
\[
    a + \cancelto{0}{(b-b)} = a
\]
\end{referencecode}

\[
    a + \cancelto{0}{(b-b)} = a
\]
