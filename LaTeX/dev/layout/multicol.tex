\packageentry{columns}{multicol -- Multiple columns}
\partlabel{Preamble}
\begin{referencecode}
\usepackage{multicol,lipsum}
\setlength{\columnsep}{8mm}
\setlength{\columnseprule}{0pt}
\end{referencecode}
\partlabel{Balanced columns}
\begin{referencecode}
\begin{multicols}{2}
    \lipsum[1][1-4]
    \columnbreak
    \lipsum[2][1-4]
\end{multicols}
\end{referencecode}
\begingroup
\setlength{\columnsep}{8mm}\setlength{\columnseprule}{0pt}

\begin{multicols}{2}
    \lipsum[1][1-4]
    \columnbreak
    \lipsum[2][1-4]
\end{multicols}
\par\endgroup
Use \texttt{\string\linewidth} to size content inside a column.
Ordinary single-column floats are not supported within \texttt{multicols}.
