
\begin{nobreaks}
\packageentry{spacedpar}{spacedpar -- Separate selected paragraphs}
\partlabel{Definition}
This template command is defined in
\texttt{Packages/Layout/package\_config.tex}; no extra package is needed.
\begin{referencecode}
\newcommand{\spacedpar}{\par\addvspace{\baselineskip}\noindent}
\end{referencecode}
\partlabel{Usage}
\begin{referencecode}
The open-loop poles determine the starting points of the root locus.
\spacedpar The asymptotes describe its unbounded branches.
\end{referencecode}

The open-loop poles determine the starting points of the root locus.
\spacedpar The asymptotes describe its unbounded branches.

\verb|\par| ends the current paragraph,
\texttt{\string\addvspace} requests one baseline of separation and
\texttt{\string\noindent} starts the next paragraph without indentation.
Adjacent \texttt{\string\addvspace} requests do not simply accumulate.
Use this command between prose paragraphs; it does not keep them on
the same page. For document-wide spacing, adjust
\texttt{\string\parskip} instead.
\end{nobreaks}
