\partlabel{Default node distance}
\verb|\dnd| (default node distance) is a length defined by the template
in \path{Packages/Graphics/package_config.tex}, with an initial value
of \texttt{1cm}. The shared \texttt{every picture} style sets
\verb|node distance=\dnd| for each new TikZ picture.
Change it with \verb|\setlength{\dnd}{12mm}| before the picture.
Without a surrounding group, this affects subsequent pictures too;
\verb|\begingroup| and \verb|\endgroup| keep the change local.
The following example uses this length through \verb|right=of ...|,
which measures the gap between node borders for these horizontal nodes.
An explicit picture option such as \verb|[node distance=8mm]| overrides
this default; \verb|right=15mm of ...| sets a distance for one node.

\partlabel{Series blocks}
\begin{referencecode}
\begingroup
\setlength{\dnd}{12mm}
\begin{tikzpicture}
    \node[input] (r) {$R(s)$};
    \node[block,right=of r] (g1) {$s+4$};
    \node[block,right=of g1] (g2) {$G(s)$};
    \node[output,right=of g2] (y) {$Y(s)$};
    \draw[->] (r) -- (g1);
    \draw[->] (g1) -- (g2);
    \draw[->] (g2) -- (y);
\end{tikzpicture}
\endgroup
\end{referencecode}

\verb|right=of g1| uses \texttt{node distance};
\verb|below=15mm of g1| sets a local distance.
\verb|(g1.south)| selects the lower anchor.
The template defines \texttt{input}, \texttt{output}, \texttt{block},
\texttt{sum}, \texttt{pickoff}, \texttt{plus} and \texttt{minus} in
\texttt{Packages/Graphics/package\_config.tex}.
